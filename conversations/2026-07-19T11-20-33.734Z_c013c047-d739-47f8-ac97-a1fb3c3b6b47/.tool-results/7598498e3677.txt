     1  #!/usr/bin/env python3
     2  """
     3  One-shot GitHub App creator via the manifest flow.
     4  
     5  Starts a local HTTP server that:
     6  1. Serves a page with a pre-filled manifest form pointing at GitHub
     7  2. Catches the OAuth callback after the user clicks "Create" on GitHub
     8  3. Exchanges the temporary code for app credentials (app_id, pem, client_secret, etc.)
     9  4. Saves everything to a JSON file and exits
    10  
    11  Usage:
    12      python3 create-github-app.py --org=ORG --name=NAME [options]
    13  
    14  Then open http://localhost:PORT in your browser (auto-opened on macOS).
    15  
    16  The user clicks "Go to GitHub", then clicks "Create GitHub App for ORG" — two clicks total.
    17  """

   409  class Handler(http.server.BaseHTTPRequestHandler):
   410      def do_GET(self):
   411          parsed = urllib.parse.urlparse(self.path)
   412  
   413          if parsed.path == "/":
   414              self._respond(200, FORM_PAGE)
   415  
   416          elif parsed.path == "/callback":
   417              params = urllib.parse.parse_qs(parsed.query)
   418              code = params.get("code", [None])[0]
   419              state = params.get("state", [None])[0]
   420  
   421              if state != STATE:
   422                  self._respond(400, error_page("State mismatch — this may be a CSRF attempt. Please try again."))
   423                  return
   424              if not code:
   425                  self._respond(400, error_page("No authorization code received from GitHub."))
   426                  return
   427  
   428              try:
   429                  url = f"https://api.github.com/app-manifests/{code}/conversions"
   430                  req = urllib.request.Request(url, method="POST", data=b"",
   431                      headers={"Accept": "application/vnd.github+json"})
   432                  resp = urllib.request.urlopen(req)
   433                  data = json.loads(resp.read().decode())
   434  
   435                  creds = {
   436                      "app_id": data["id"],
   437                      "app_slug": data.get("slug"),
   438                      "client_id": data.get("client_id"),
   439                      "client_secret": data.get("client_secret"),
   440                      "pem": data.get("pem"),
   441                      "webhook_secret": data.get("webhook_secret"),
   442                      "html_url": data.get("html_url"),
   443                      "owner": data.get("owner", {}).get("login"),
   444                  }
   445  
   446                  # Write credentials with restricted permissions (contains private key)
   447                  fd = os.open(args.output, os.O_WRONLY | os.O_CREAT | os.O_TRUNC, 0o600)
   448                  with os.fdopen(fd, "w") as f:
   449                      json.dump(creds, f, indent=2)
   450  
   451                  self._respond(200, success_page(
   452                      name=args.name,
   453                      app_id=data["id"],
   454                      slug=data.get("slug", "unknown"),
   455                      html_url=data.get("html_url", ""),
   456                      output_path=os.path.abspath(args.output)
   457                  ))
   458  
   459                  print(f"\n✅ App created! ID: {data['id']}, Slug: {data.get('slug')}")
   460                  print(f"   Credentials: {os.path.abspath(args.output)}")
   461  
   462                  threading.Thread(target=lambda: (
   463                      time.sleep(1),
   464                      os._exit(0)
   465                  ), daemon=True).start()
   466  
   467              except urllib.error.HTTPError as e:
   468                  body = e.read().decode()
   469                  self._respond(500, error_page(body))
   470                  print(f"❌ Error: {body}", file=sys.stderr)
   471          else:
   472              self._respond(404, error_page("Page not found."))
   473  
   474      def _respond(self, status, body):
   475          self.send_response(status)
   476          self.send_header("Content-Type", "text/html; charset=utf-8")
   477          self.end_headers()
   478          self.wfile.write(body.encode())
   479  
   480      def log_message(self, format, *args):
   481          pass

   484  def is_port_available(port):
   485      with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:
   486          try:
   487              s.bind(("127.0.0.1", port))
   488              return True
   489          except OSError:
   490              return False
   491  
   492  
   493  if __name__ == "__main__":
   494      if not is_port_available(PORT):
   495          print(f"❌ Port {PORT} is already in use. Try --port=DIFFERENT_PORT", file=sys.stderr)
   496          sys.exit(1)
   497  
   498      print(f"🌐 Open http://localhost:{PORT} in your browser")
   499      print(f"   Waiting for GitHub App creation...")
   500  
   501      if not args.no_open and sys.platform == "darwin":
   502          os.system(f"open http://localhost:{PORT}")
   503  
   504      server = http.server.HTTPServer(("127.0.0.1", PORT), Handler)
   505      try:
   506          server.serve_forever()
   507      except KeyboardInterrupt:
   508          print("\nCancelled")
   509          server.server_close()
